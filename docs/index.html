import { calcularRuta, RespuestaRutaCompleta } from 'ruta-craft-osm';

// Cálculo de ruta óptima multi-parada con C++ nativo
const respuesta: RespuestaRutaCompleta = await calcularRuta({
  puntos: [
    [29.065639, -110.056448], // Origen (A)
    [29.062039, -110.059091], // Parada 1
    [29.059486, -110.053973], // Parada 2
    [29.118091, -109.970340]  // Destino (B)
  ],
  cachePath: './cache_rutas.json'
});

console.log(`⚡ Distancia: ${respuesta.resultado.distancia_total_km} km`);
console.log(`📍 Puntos polilínea: ${respuesta.resultado.ruta_coordenadas.length}`);
console.log(respuesta.resultado.instrucciones[0]); // { accion: "Sigue recto", calle: "...", distancia_m: 466.3 }
const { calcularRuta } = require('ruta-craft-osm');

// Compatible con procesos Main y Renderer (Preload) en Electron
async function obtenerTrazado() {
  const respuesta = await calcularRuta({
    puntos: [
      [29.065639, -110.056448],
      [29.118091, -109.970340]
    ]
  });

  if (respuesta.exito) {
    // Renderizar en Leaflet / Mapbox
    dibujarRutaEnMapa(respuesta.resultado.ruta_coordenadas);
  }
}
