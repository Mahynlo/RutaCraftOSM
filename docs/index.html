Usa el motor de rutas desde TypeScript, Electron o directamente desde la terminal

import { calcularRuta, RespuestaRutaCompleta } from 'ruta-craft-osm';

// Cálculo de ruta multi-parada con el motor C++ nativo
const respuesta: RespuestaRutaCompleta = await calcularRuta({
  puntos: [
    [29.065639, -110.056448], // Origen (A)
    [29.062039, -110.059091], // Parada 1
    [29.059486, -110.053973], // Parada 2
    [29.118091, -109.970340]  // Destino (B)
  ]
});

console.log(`Distancia: ${respuesta.resultado.distancia_total_km} km`);
console.log(`Puntos polilínea: ${respuesta.resultado.ruta.length}`);
console.log(respuesta.resultado.instrucciones[0]); // { accion: "Sigue recto", calle: "...", distancia_m: ... }
const { app, ipcMain } = require('electron');
const path = require('path');
const { crearMotorRutas } = require('ruta-craft-osm');

// Proceso main: un solo motor con el grafo cargado en memoria (< 1 ms por ruta)
const motor = crearMotorRutas({
  cachePath: path.join(app.getPath('userData'), 'cache_rutas.json')
});

ipcMain.handle('calcular-ruta', (_evento, puntos) => motor.calcular(puntos));

// Guardar el caché al salir
app.on('before-quit', () => motor.cerrar());
# Desde ruta-craft-osm/bin/cli/ (el grafo está en grafos/ junto al ejecutable)
.\cli.exe --array 29.065639 -110.056448 29.059486 -110.053973 --sin-cache

# Puntos desde un archivo JSON y resultado en otro archivo
.\cli.exe --puntos paradas.json --salida ruta.json

# Modo servidor: una petición JSON por línea en stdin
'{"id":1,"puntos":[[29.065639,-110.056448],[29.059486,-110.053973]]}' | .\cli.exe --serve --sin-cache
